# Локальный синтез речи, лёгкий путь (движок kokoro-ru, ONNX).
#
# Работает на любом CPU, CUDA не нужна, PyTorch не нужен.
# Проверено на этом проекте: RTF ~0.31 на CPU (в 3 раза быстрее реального
# времени), поэтому kokoro — движок по умолчанию там, где нет NVIDIA.
#
# При первом запуске НЕ ставится. Догружается с полным режимом
# (`full_deps.py` / кнопка в «О лисе»). Без этих пакетов ассистент
# озвучивает реплики через edge-tts, а затем через SAPI5.
#
# ВАЖНО про размер: кроме модели синтеза (~330 МБ) нужен ещё текстовый
# фронтенд. ruaccent скачивает свою модель ударений с huggingface.co в момент
# построения движка и складывает её внутрь своего пакета — это ещё ~690 МБ.
# Поэтому без доступа к huggingface.co kokoro не поднимется, а `tts_local.py
# kokoro` при подготовке машины заранее качает и её.
#
# Установка вручную:
#     .venv\Scripts\python.exe -m pip install -r requirements-tts.txt
#     .venv\Scripts\python.exe tts_local.py kokoro      # скачать модель
#
# Версии зафиксированы по проверенному рабочему окружению (.venv).

# Инференс и загрузка модели
onnxruntime==1.30.0
# ВАЖНО: не 2.x. Транзитивно через ruaccent тянется transformers, который
# требует huggingface-hub<2.0,>=1.5.0 — с 2.0.0 установка встанет конфликтом.
huggingface_hub==1.33.0

# Фронтенд текста -> IPA: расстановка ударений, фонемизация через eSpeak
ruaccent==1.5.8.3
misaki==0.9.4
phonemizer==3.4.0
espeakng-loader==0.2.4
